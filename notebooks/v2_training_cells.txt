import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.utils.class_weight import compute_class_weight
from tensorflow import keras
from tensorflow.keras import layers, callbacks
import time

# 1. Carve a validation set out of train (stratified) — test stays fully untouched
X_tr, X_val, y_tr, y_val = train_test_split(
    X_train_seq, y_train_seq, test_size=0.1, stratify=y_train_seq, random_state=42
)
print("Train:", X_tr.shape, " Val:", X_val.shape)

# 2. Class weights — balanced, but capped to avoid instability from classes
# with single-digit sample counts (Heartbleed=8, Sql Injection=16). Uncapped
# balanced weighting would give these classes weights in the thousands, which
# can destabilize training without meaningfully helping — the model genuinely
# can't learn a robust boundary from 8 examples no matter how it's weighted.
classes = np.unique(y_tr)
raw_weights = compute_class_weight('balanced', classes=classes, y=y_tr)
capped_weights = np.clip(raw_weights, None, 50)
class_weight_dict = dict(zip(classes.astype(int), capped_weights))
print("Class weights (capped at 50):", class_weight_dict)

# 3. Model — same architecture verified earlier this session (83,919 params)
model = keras.Sequential([
    layers.Conv1D(64, 3, activation='relu', input_shape=(10, 20)),
    layers.BatchNormalization(),
    layers.MaxPooling1D(2),
    layers.Conv1D(128, 3, activation='relu', padding='same'),
    layers.BatchNormalization(),
    layers.MaxPooling1D(2),
    layers.LSTM(64),
    layers.Dense(64, activation='relu'),
    layers.Dense(15, activation='softmax')
])
model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
model.summary()

# 4. Timed sanity run — 2 epochs, to get a real per-epoch estimate before committing further
start = time.time()
history = model.fit(
    X_tr, y_tr,
    validation_data=(X_val, y_val),
    epochs=2,
    batch_size=512,
    class_weight=class_weight_dict,
    verbose=1
)
elapsed = time.time() - start
print(f"\n2 epochs took {elapsed:.1f}s ({elapsed/2:.1f}s/epoch on full {len(X_tr):,}-sequence train set)")

    return np.concatenate(X_parts, axis=0), np.concatenate(y_parts, axis=0)
